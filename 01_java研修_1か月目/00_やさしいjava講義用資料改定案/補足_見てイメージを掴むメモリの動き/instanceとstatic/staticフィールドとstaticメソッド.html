Java Memory: Pure Static Roles (Utility vs Field)

class MyUtils {
static int callCount = 0;
static int add(int a, int b) {
return a + b; // フィールド不使用
}
static void record() {
callCount++; // フィールドを更新
}
}
public class Main {
public static void main(String[] args) {
int sum = MyUtils.add(10, 20);
MyUtils.record();
System.out.println("sum:" + sum + ", count:" + MyUtils.callCount);
}
}
