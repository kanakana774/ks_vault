Java Memory: Static Method vs Instance Method

class Robot {
static void sayHello() { ... }
void sayName() { ... }
}
public class Main {
public static void main(String[] args) {
Robot.sayHello(); // static呼び出し
Robot rb = null;
rb.sayName(); // エラー発生！
rb = new Robot();
rb.sayName(); // 成功！
}
}
