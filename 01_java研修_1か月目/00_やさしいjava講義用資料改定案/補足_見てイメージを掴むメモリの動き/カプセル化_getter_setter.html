Java Memory: Encapsulation (Getter & Setter)

class Robot {
private int power = 0;
public void setPower(int p) {
this.power = p;
}
public int getPower() {
return this.power;
}
}
public class Main {
public static void main(String[] args) {
Robot rb = new Robot();
rb.setPower(80);
int val = rb.getPower();
System.out.println(val);
}
}
