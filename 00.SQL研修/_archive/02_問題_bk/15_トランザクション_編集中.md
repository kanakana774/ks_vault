### 問題1：
顧客 Charlie（customer_id=3）のメールアドレスを charlie@example.com に変更し、
その更新を ROLLBACK して元に戻してください。
ただし、ROLLBACK前にSELECT文を実行して、データの更新がされていることと
ROLLBACK後にSELECT文を実行して、データが元に戻っていることを確認してください。

```SQL
BEGIN;

UPDATE customers
SET
	email = 'charlie@example.com'
WHERE
	customer_id = 3;

-- ここで確認すると更新されている						
SELECT
	*
FROM
	customers
WHERE
	customer_id = 3;

ROLLBACK;

-- ここで確認 → 元の状態に戻っている						
SELECT
	*
FROM
	customers
WHERE
	customer_id = 3;
```

### 問題2：
新しい顧客 Frances を登録し、そのINSERTを COMMIT してください。
(customer_id：6、name：Frances、email：frances@example.com、
phone：090-6756-6655、city：Kyoto、join_date：2022-02-01、membership_id：1)
ただし、COMMIT前にSELECT文を実行して、データの登録されていること
COMMIT後にもSELECT文を実行して、変わらずデータが登録されていることを確認してください。
```SQL
BEGIN;

INSERT INTO
	customers (
		customer_id,
		name,
		email,
		phone,
		city,
		join_date,
		membership_id
	)
VALUES
	(
		6,
		'Frances',
		'frances@example.com',
		'090-6756-6655',
		'Kyoto',
		'2022-02-01',
		1
	);

-- ここで確認 → データは登録されている					
SELECT
	*

	customers

-- ここで確認 → COMMIT 済みなのでデータは残っている

	*
FROM
	customers
WHERE
	name = 'Frances';
```

### 問題3：
顧客 Bob（customer_id=2）の電話番号を090-2222-9999に更新するトランザクションを実行してください。	
その後、セーブポイントを作り、顧客Bobの更新後の状態まで、戻れるようにしてください。	
セーブポイント作成後に顧客Alice(customer_id=1)の電話番号を000-0000-0000に更新してください。	
SELECT文を実行し、2人のデータが更新されていることを確認してください。	
顧客Aliceのデータが更新はミスであるため、セーブポイントまでROLLBACKをしてください。	
その後COMMITを行い、顧客Bobのデータのみ更新されていることを確認してください。	
```SQL
BEGIN;

-- 正しい更新(Bobの電話番号変更)					
UPDATE customers
SET
	phone = '090-2222-9999'
WHERE
	customer_id = 2;

SAVEPOINT sp1;

-- 誤って Alice の電話番号も変更					
UPDATE customers
SET
	phone = '000-0000-0000'
WHERE
	customer_id = 1;

-- ここで確認 → 2人ともデータが更新されている					
SELECT
	*
FROM
	customers
WHERE
	customer_id IN (1, 2);

-- 間違いを取り消す					
ROLLBACK TO SAVEPOINT sp1;

-- Bob のみ変更が残る					
COMMIT;

-- ここで確認 → Bobだけデータが更新されている					
SELECT
	*
FROM
	customers
WHERE
	customer_id IN (1, 2);
```

### 問題4：
以下の処理をトランザクションとPL/pgSQLを用いて行ってください。
顧客 Alice（customer_id=1）から、注文(order_id=102)が入りました。
注文日時はシステム時刻、ステータスはPendingとして注文登録をしてください。
注文された商品はLaptop(ノートパソコン)が2つです。
上記をもとに注文明細の登録を行ってください。
その後、対応する商品の在庫数を減らす処理を行ってください。
在庫数が必要十分以上あるのであればCOMMITを行ってください。
ただし、在庫数がマイナスになる場合は注文をキャンセル（ROLLBACK）してください。

```SQL
CREATE OR REPLACE PROCEDURE order_process (
	p_product_name varchar,
	p_order_quantity integer,
	p_customer_id integer
) AS $$
DECLARE
v_order_id integer;
v_product_id integer;
v_stock_quantity integer;
v_order_item_id integer;

BEGIN
	-- 商品名から商品IDを特定
	SELECT
		product_id INTO v_product_id
	FROM
		products
	WHERE
		product_name = p_product_name;

	-- 最新の注文IDの
	SELECT
		MAX(order_id) + 1 INTO v_order_id
	FROM
		orders;

	INSERT INTO
		orders
	VALUES
		(
			v_order_id,
			p_customer_id,
			current_timestamp,
			'pending'
		);

	SELECT
		MAX(order_item_id) + 1 INTO v_order_item_id
	FROM
		orderitems;

	INSERT INTO
		orderitems (order_item_id, order_id, product_id, quantity)
	VALUES
		(
			v_order_item_id,
			v_order_id,
			v_product_id,
			p_order_quantity
		);

	raise notice '%',v_product_id;

	UPDATE inventory
	SET
		stock_quantity = stock_quantity - p_order_quantity
	WHERE
		product_id = v_product_id
		AND warehouse_id = 1
	RETURNING
		stock_quantity INTO v_stock_quantity;

	if v_stock_quantity >= 0 THEN
		COMMIT;
	ELSE
		ROLLBACK;
		raise notice '在庫がありません。 %',v_stock_quantity;
	END if;
END;
$$ language plpgsql;
```
