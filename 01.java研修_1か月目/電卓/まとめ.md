正規表現を一切使わず、すべてのロジックで「マイナス記号の判定（引き算 vs 負の数）」を実装

コードが長くなりますが、コピー＆ペーストでそのまま動くようになっています。mainメソッドで全てのパターンを実行し比較します。

### ソースコード完全版

```java
import java.util.*;
import javax.script.ScriptEngine;
import javax.script.ScriptEngineManager;

public class UltimateCalculator {

    public static void main(String[] args) {
        // テスト用計算式
        // 4 * -5 = -20
        // 10 - -3 = 13
        // -20 + 13 = -7
        // ※パターンA以外はカッコ()にも対応しています
        String expression = "4 * -5 + 10 - -3"; 
        
        System.out.println("計算対象: " + expression);
        System.out.println("=========================================");

        // 1. A. リスト分割法
        try {
            System.out.println("【A. リスト分割法】");
            double result = ListSplitCalculator.calc(expression);
            System.out.println("結果: " + result);
        } catch (Exception e) { System.out.println("エラー: " + e.getMessage()); }
        System.out.println("-----------------------------------------");

        // 2. B. 2つのスタック法 (Deque)
        try {
            System.out.println("【B. 2つのスタック法】");
            double result = StackCalculator.calc(expression);
            System.out.println("結果: " + result);
        } catch (Exception e) { System.out.println("エラー: " + e.getMessage()); }
        System.out.println("-----------------------------------------");

        // 3. C. 再帰下降構文解析 (Recursive Descent)
        try {
            System.out.println("【C. 再帰下降構文解析】");
            double result = RecursiveCalculator.calc(expression);
            System.out.println("結果: " + result);
        } catch (Exception e) { System.out.println("エラー: " + e.getMessage()); }
        System.out.println("-----------------------------------------");

        // 4. D. 逆ポーランド記法 (RPN)
        try {
            System.out.println("【D. 逆ポーランド記法】");
            double result = RpnCalculator.calc(expression);
            System.out.println("結果: " + result);
        } catch (Exception e) { System.out.println("エラー: " + e.getMessage()); }
        System.out.println("-----------------------------------------");

        // 5. 番外. ScriptEngine
        try {
            System.out.println("【番外. ScriptEngine】");
            Object result = ScriptEngineCalc.calc(expression);
            System.out.println("結果: " + result);
        } catch (Exception e) { System.out.println("エラー: " + e.getMessage()); }
        System.out.println("=========================================");
    }

    // =================================================================
    // 【A】リスト分割法 (あなたの考案したロジック)
    // =================================================================
    static class ListSplitCalculator {
        public static double calc(String expression) {
            List<Double> numbers = new ArrayList<>();
            List<Character> operators = new ArrayList<>();
            StringBuilder currentNumber = new StringBuilder();

            for (int i = 0; i < expression.length(); i++) {
                char c = expression.charAt(i);
                if (c == ' ') continue;

                if (Character.isDigit(c) || c == '.') {
                    currentNumber.append(c);
                } else if (isOperator(c)) {
                    // ★判定: バッファが空なら負の符号
                    if (c == '-' && currentNumber.length() == 0) {
                        currentNumber.append(c);
                    } else {
                        if (currentNumber.length() > 0) {
                            numbers.add(Double.parseDouble(currentNumber.toString()));
                            currentNumber.setLength(0);
                        }
                        operators.add(c);
                    }
                }
            }
            if (currentNumber.length() > 0) {
                numbers.add(Double.parseDouble(currentNumber.toString()));
            }

            // 乗除
            int i = 0;
            while (i < operators.size()) {
                char op = operators.get(i);
                if (op == '*' || op == '/') {
                    double v1 = numbers.get(i);
                    double v2 = numbers.get(i + 1);
                    numbers.set(i, (op == '*') ? v1 * v2 : v1 / v2);
                    numbers.remove(i + 1);
                    operators.remove(i);
                } else {
                    i++;
                }
            }
            // 加減
            double res = numbers.get(0);
            for (int j = 0; j < operators.size(); j++) {
                char op = operators.get(j);
                double next = numbers.get(j + 1);
                if (op == '+') res += next;
                else if (op == '-') res -= next;
            }
            return res;
        }
        private static boolean isOperator(char c) {
            return c == '+' || c == '-' || c == '*' || c == '/';
        }
    }

    // =================================================================
    // 【B】2つのスタック法 (Deque使用) - Dijkstra's Two-Stack Algorithm
    // =================================================================
    static class StackCalculator {
        public static double calc(String expression) {
            Deque<Double> values = new ArrayDeque<>();
            Deque<Character> ops = new ArrayDeque<>();

            for (int i = 0; i < expression.length(); i++) {
                char c = expression.charAt(i);
                if (c == ' ') continue;

                // ★判定: 先頭 or 直前が数字じゃない場合の '-' は負号
                boolean isNegativeSign = (c == '-') && 
                    (i == 0 || (!Character.isDigit(expression.charAt(i - 1)) && expression.charAt(i - 1) != ')'));

                if (Character.isDigit(c) || isNegativeSign) {
                    StringBuilder sb = new StringBuilder();
                    if (isNegativeSign) {
                        sb.append('-'); i++;
                    }
                    while (i < expression.length() && (Character.isDigit(expression.charAt(i)) || expression.charAt(i) == '.')) {
                        sb.append(expression.charAt(i)); i++;
                    }
                    values.push(Double.parseDouble(sb.toString()));
                    i--;
                } else if (c == '(') {
                    ops.push(c);
                } else if (c == ')') {
                    while (ops.peek() != '(') {
                        values.push(applyOp(ops.pop(), values.pop(), values.pop()));
                    }
                    ops.pop();
                } else if ("+-*/".indexOf(c) != -1) {
                    while (!ops.isEmpty() && hasPrecedence(c, ops.peek())) {
                        values.push(applyOp(ops.pop(), values.pop(), values.pop()));
                    }
                    ops.push(c);
                }
            }
            while (!ops.isEmpty()) {
                values.push(applyOp(ops.pop(), values.pop(), values.pop()));
            }
            return values.pop();
        }

        private static boolean hasPrecedence(char op1, char op2) {
            if (op2 == '(' || op2 == ')') return false;
            if ((op1 == '*' || op1 == '/') && (op2 == '+' || op2 == '-')) return false;
            return true;
        }

        private static double applyOp(char op, double b, double a) {
            switch (op) {
                case '+': return a + b;
                case '-': return a - b;
                case '*': return a * b;
                case '/': return a / b;
            }
            return 0;
        }
    }

    // =================================================================
    // 【C】再帰下降構文解析 (Recursive Descent Parser)
    // =================================================================
    static class RecursiveCalculator {
        private String text;
        private int pos = 0;

        public static double calc(String expression) {
            RecursiveCalculator parser = new RecursiveCalculator();
            parser.text = expression.replace(" ", "");
            parser.pos = 0;
            return parser.parseExpression();
        }

        // 足し算・引き算
        private double parseExpression() {
            double v = parseTerm();
            while (pos < text.length()) {
                char op = text.charAt(pos);
                if (op != '+' && op != '-') break;
                pos++;
                double v2 = parseTerm();
                if (op == '+') v += v2; else v -= v2;
            }
            return v;
        }

        // 掛け算・割り算
        private double parseTerm() {
            double v = parseFactor();
            while (pos < text.length()) {
                char op = text.charAt(pos);
                if (op != '*' && op != '/') break;
                pos++;
                double v2 = parseFactor();
                if (op == '*') v *= v2; else v /= v2;
            }
            return v;
        }

        // 数字・カッコ・負の数
        private double parseFactor() {
            if (pos >= text.length()) return 0;
            char c = text.charAt(pos);

            // ★判定: ここで '-' が来たら再帰で符号反転
            if (c == '-') {
                pos++;
                return -parseFactor();
            }
            if (c == '(') {
                pos++;
                double v = parseExpression();
                pos++; // ')' を飛ばす
                return v;
            }

            StringBuilder sb = new StringBuilder();
            while (pos < text.length()) {
                c = text.charAt(pos);
                if (Character.isDigit(c) || c == '.') {
                    sb.append(c);
                    pos++;
                } else break;
            }
            return Double.parseDouble(sb.toString());
        }
    }

    // =================================================================
    // 【D】逆ポーランド記法 (RPN + Shunting-yard)
    // =================================================================
    static class RpnCalculator {
        public static double calc(String expression) {
            return evaluateRPN(infixToRPN(expression));
        }

        // 中置記法 -> RPN
        private static List<String> infixToRPN(String expression) {
            List<String> output = new ArrayList<>();
            Deque<Character> stack = new ArrayDeque<>();

            for (int i = 0; i < expression.length(); i++) {
                char c = expression.charAt(i);
                if (c == ' ') continue;

                // ★判定: スタックBと同様のロジック
                boolean isNegativeSign = (c == '-') && 
                    (i == 0 || (!Character.isDigit(expression.charAt(i - 1)) && expression.charAt(i - 1) != ')'));

                if (Character.isDigit(c) || isNegativeSign) {
                    StringBuilder sb = new StringBuilder();
                    if (isNegativeSign) { sb.append('-'); i++; }
                    while (i < expression.length() && (Character.isDigit(expression.charAt(i)) || expression.charAt(i) == '.')) {
                        sb.append(expression.charAt(i)); i++;
                    }
                    output.add(sb.toString());
                    i--;
                } else if (c == '(') {
                    stack.push(c);
                } else if (c == ')') {
                    while (!stack.isEmpty() && stack.peek() != '(') output.add(String.valueOf(stack.pop()));
                    stack.pop();
                } else if ("+-*/".indexOf(c) != -1) {
                    while (!stack.isEmpty() && getPrecedence(stack.peek()) >= getPrecedence(c)) {
                        output.add(String.valueOf(stack.pop()));
                    }
                    stack.push(c);
                }
            }
            while (!stack.isEmpty()) output.add(String.valueOf(stack.pop()));
            return output;
        }

        // RPN計算
        private static double evaluateRPN(List<String> tokens) {
            Deque<Double> stack = new ArrayDeque<>();
            for (String t : tokens) {
                if ("+-*/".contains(t) && t.length() == 1) {
                    double b = stack.pop();
                    double a = stack.pop();
                    switch (t.charAt(0)) {
                        case '+': stack.push(a + b); break;
                        case '-': stack.push(a - b); break;
                        case '*': stack.push(a * b); break;
                        case '/': stack.push(a / b); break;
                    }
                } else {
                    stack.push(Double.parseDouble(t));
                }
            }
            return stack.pop();
        }

        private static int getPrecedence(char op) {
            if (op == '*' || op == '/') return 2;
            if (op == '+' || op == '-') return 1;
            return 0;
        }
    }

    // =================================================================
    // 【番外】ScriptEngine
    // =================================================================
    static class ScriptEngineCalc {
        public static Object calc(String expression) throws Exception {
            ScriptEngineManager manager = new ScriptEngineManager();
            ScriptEngine engine = manager.getEngineByName("JavaScript");
            if (engine == null) return "JS Engine not found (Java15+)";
            return engine.eval(expression);
        }
    }
}
```

### 全パターンの詳細比較

最後に、各アプローチのメリット・デメリットを整理します。

| パターン | 名称 | マイナス判定のロジック | メリット | デメリット |
| :--- | :--- | :--- | :--- | :--- |
| **A** | **リスト分割法** | バッファ(`currentNumber`)が空なら負号 | **直感的で分かりやすい**。<br>コードの流れが追いやすい。 | 処理効率がやや悪い（リストを何度も走査）。<br>カッコ `()` 対応が困難。 |
| **B** | **2つのスタック法** | 直前が「数字以外」または「先頭」なら負号 | **実用的かつ効率的**。<br>ループ1回で計算完了。<br>カッコ対応も容易。 | スタック操作の理解が必要。<br>優先順位判定が少し複雑。 |
| **C** | **再帰下降構文解析** | `factor`メソッド内で`-`が来たら再帰 | **構造が最も美しい**。<br>「掛け算は足し算より先」がコードの形そのもの。<br>拡張性が最強。 | コードが長くなり、ファイルが縦に伸びる。<br>再帰の概念が必要。 |
| **D** | **逆ポーランド記法** | スタック法と同じ | コンパイラ等で使われる**最速の手法**。<br>計算ロジックが単純明快。 | 実装が「変換」と「計算」の2段階になり面倒。<br>人間には直感的でない。 |
| **番外** | **ScriptEngine** | JSエンジンにお任せ | **実装時間ほぼゼロ**。<br>関数なども使える。 | Javaのバージョン依存がある。<br>学習目的にはならない。 |

### どれを使うべき？

*   **学習用・単純な計算:**
    今回あなたが考案した **【A. リスト分割法】** は、正規表現を使わずに文字列操作の基礎を学ぶ上で非常に優れたアプローチです。

*   **一般的なアプリ開発:**
    **【B. 2つのスタック法】** がバランスが良く、標準的な選択肢です。

*   **言語自作・高度な解析:**
    **【C. 再帰下降構文解析】** や **【D. 逆ポーランド記法】** が使われます。
